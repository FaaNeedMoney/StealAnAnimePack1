-- ── CHAR REFRESH ──────────────────────────────────────────────
local function refresh_char(c)
    Char = c
    HRP  = c:WaitForChild("HumanoidRootPart")
    Hum  = c:WaitForChild("Humanoid")
    if not State.Base then State.Base = HRP.CFrame end
    if not State.FarmSpot then State.FarmSpot = HRP.CFrame end
    task.wait(0.5)
    Hum.WalkSpeed = CFG.StealSpeed
    Hum.JumpPower = 0
    pcall(function() HRP:SetNetworkOwner(nil) end)
end
LP.CharacterAdded:Connect(refresh_char)

local function set_speed(v)
    CFG.StealSpeed = math.clamp(v, 16, 2000)
    Hum.WalkSpeed  = CFG.StealSpeed
end

-- ── REMOTE HELPER ─────────────────────────────────────────────
local function fire_remote(name, ...)
    local r = ReplicatedStorage:FindFirstChild(name, true)
           or Workspace:FindFirstChild(name, true)
    if not r then return end
    if     r:IsA("RemoteEvent")    then r:FireServer(...)
    elseif r:IsA("RemoteFunction") then return r:InvokeServer(...) end
end

-- ══════════════════════════════════════════════════════════════
--  MOVEMENT — tween + AC bypass hop
--  why hop: "Steal an Anime Pack" detects large CFrame deltas
--  and teleports you back; small hops (<= HopDist) stay under
--  the detection threshold. Each hop is a tween so it looks
--  like legit movement to the velocity check.
-- ══════════════════════════════════════════════════════════════
local function cancel_tween()
    if State.ActiveTween then
        pcall(function() State.ActiveTween:Cancel() end)
        State.ActiveTween = nil
    end
end

local function do_tween(cf, t)
    cancel_tween()
    local tw = TweenService:Create(
        HRP,
        TweenInfo.new(t or 0.05, Enum.EasingStyle.Linear, Enum.EasingDirection.Out),
        {CFrame = cf}
    )
    State.ActiveTween = tw
    tw:Play()
    tw.Completed:Wait()
    if State.ActiveTween == tw then State.ActiveTween = nil end
end

local function move_to(targetCF)
    local origin = HRP.CFrame
    local dist   = (targetCF.Position - origin.Position).Magnitude

    if dist < 3 then
        do_tween(targetCF, 0.05)
        return
    end

    if not CFG.ACBypass or dist <= CFG.HopDist then
        do_tween(targetCF, math.clamp(dist / 200, 0.05, 0.4))
        return
    end

    -- hop in CFG.HopDist chunks, each a tween
    local steps = math.ceil(dist / CFG.HopDist)
    for i = 1, steps do
        if not (State.Stealing or State.Placing) then break end
        local hop = origin:Lerp(targetCF, i / steps)
        do_tween(hop, CFG.HopTime)
        -- if server kicked us back, snap to hop and retry once
        local drift = (HRP.CFrame.Position - hop.Position).Magnitude
        if drift > 16 then
            task.wait(0.08)
            cancel_tween()
            HRP.CFrame = hop
            task.wait(0.03)
        end
    end

    -- final snap to guarantee arrival
    cancel_tween()
    HRP.CFrame = targetCF
end

-- ══════════════════════════════════════════════════════════════
--  TIER DETECTION
-- ══════════════════════════════════════════════════════════════
local function str_has_tier(str)
    local low = str:lower()
    for tier in pairs(CFG.Tiers) do
        if low:find(tier, 1, true) then return true end
    end
    return false
end

local function obj_matches_tier(obj)
    if str_has_tier(obj.Name) then return true end
    for _, ch in ipairs(obj:GetChildren()) do
        if ch:IsA("StringValue") or ch:IsA("TextLabel") or ch:IsA("TextBox") then
            local v = ch:IsA("StringValue") and ch.Value or ch.Text
            if v and str_has_tier(v) then return true end
        end
    end
    return false
end

local function get_targets()
    local out = {}
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if (obj:IsA("Model") or obj:IsA("BasePart")) and obj_matches_tier(obj) then
            local part = obj:IsA("BasePart") and obj
                      or obj:FindFirstChildWhichIsA("BasePart")
            if part then
                table.insert(out, {model = obj, part = part})
            end
        end
    end
    return out
end

local function nearest_target()
    local best, bd = nil, math.huge
    for _, t in ipairs(get_targets()) do
        local d = (t.part.Position - HRP.Position).Magnitude
        if d < bd then bd = d; best = t end
    end
    return best
end

-- ══════════════════════════════════════════════════════════════
--  BASE FIND
-- ══════════════════════════════════════════════════════════════
local function get_base_cf()
    if State.Base then return State.Base end
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local n  = obj.Name:lower()
            local ok = n:find("base",1,true) or n:find("plot",1,true) or n:find("spawn",1,true)
            if ok then
                local ow = obj:FindFirstChild("Owner") or obj:FindFirstChild("PlayerName")
                if ow and (tostring(ow.Value) == LP.Name
                        or tostring(ow.Value) == tostring(LP.UserId)) then
                    return obj.CFrame + Vector3.new(0, 5, 0)
                end
            end
        end
    end
    return HRP.CFrame
end

-- ══════════════════════════════════════════════════════════════
--  GUI SKIP & BUTTON FIRE
-- ══════════════════════════════════════════════════════════════
local GUI_ROOTS = {"Fluent", "LarpGUI"}

local function is_own_gui(obj)
    local p = obj
    while p do
        for _, n in ipairs(GUI_ROOTS) do
            if p.Name == n then return true end
        end
        p = p.Parent
    end
    return false
end

local function fire_btn(btn)
    pcall(function()
        for _, c in ipairs(getconnections(btn.MouseButton1Click)) do
            if c.Function then c.Function() end
        end
    end)
    pcall(function() firesignal(btn.MouseButton1Click) end)
    pcall(function()
        for _, c in ipairs(getconnections(btn.Activated)) do
            if c.Function then c.Function() end
        end
    end)
    pcall(function() firesignal(btn.Activated) end)
    pcall(function() btn.MouseButton1Click:Fire() end)
end

local function btn_on_screen(btn)
    local ap  = btn.AbsolutePosition
    local asz = btn.AbsoluteSize
    if asz.X <= 0 or asz.Y <= 0 then return false end
    local vp  = Workspace.CurrentCamera.ViewportSize
    return ap.X < vp.X and ap.Y < vp.Y and (ap.X+asz.X) > 0 and (ap.Y+asz.Y) > 0
end

local function scan_fire(patterns)
    local gui = LP:FindFirstChildOfClass("PlayerGui")
    if not gui then return end
    for _, obj in ipairs(gui:GetDescendants()) do
        if (obj:IsA("TextButton") or obj:IsA("ImageButton"))
            and obj.Visible
            and not is_own_gui(obj)
            and btn_on_screen(obj)
        then
            local full = obj.Name:lower()
                      .. (obj:IsA("TextButton") and obj.Text:lower() or "")
            for _, pat in ipairs(patterns) do
                if full:find(pat, 1, true) then
                    fire_btn(obj)
                    task.wait(0.015)
                    break
                end
            end
        end
    end
end

-- ══════════════════════════════════════════════════════════════
--  AUTO STEAL
--  1. scan workspace for pack matching selected rarities
--  2. tween directly to the pack (with AC hop bypass)
--  3. fire steal remotes + touch trigger
--  4. tween back to base (with AC hop bypass)
-- ══════════════════════════════════════════════════════════════
local STEAL_REMOTES = {
    "StealPack", "Steal", "GrabPack", "PickupPack", "TakePack"
}

local steal_busy = false

local function do_one_steal(target)
    if steal_busy then return end
    steal_busy    = true
    Hum.WalkSpeed = CFG.StealSpeed
    pcall(function() HRP:SetNetworkOwner(nil) end)

    -- go to pack
    move_to(target.part.CFrame + Vector3.new(0, 3, 0))

    -- fire all steal remotes
    for _, name in ipairs(STEAL_REMOTES) do
        fire_remote(name, target.model)
    end

    -- touch trigger fallback
    cancel_tween()
    HRP.CFrame = target.part.CFrame + Vector3.new(0, 2, 0)
    task.wait(0.06)

    -- return to base via tween + AC bypass
    local baseCF = get_base_cf()
    move_to(baseCF)
    cancel_tween()
    HRP.CFrame = baseCF
    task.wait(0.08)

    steal_busy = false
end

local function auto_steal_loop()
    while State.Stealing do
        local t = nearest_target()
        if t and not steal_busy then
            do_one_steal(t)
        else
            task.wait(0.5)
        end
        task.wait(0.03)
    end
end

-- ══════════════════════════════════════════════════════════════
--  AUTO FARM — combo x2–x5 speed training
-- ══════════════════════════════════════════════════════════════
local COMBO_REMOTES = {
    "SpeedCombo","Combo","TrainingCombo","ClickTrain","SpeedTrain","PunchTrain"
}
local COMBO_PATS = {
    "x2","x3","x4","x5","2x","3x","4x","5x",
    "combo","train","click","speed","tap","punch","hit"
}

local function auto_farm_loop()
    while State.Farming do
        scan_fire(COMBO_PATS)
        for _, name in ipairs(COMBO_REMOTES) do fire_remote(name) end
        task.wait(0.008)
    end
end

-- ══════════════════════════════════════════════════════════════
--  AUTO HATCH
-- ══════════════════════════════════════════════════════════════
local HATCH_REMOTES = {"OpenPack","HatchPack","OpenEgg","Hatch"}
local HATCH_PATS    = {"open","hatch","ready","crack"}

local function auto_hatch_loop()
    while State.Hatching do
        for _, name in ipairs(HATCH_REMOTES) do fire_remote(name) end
        scan_fire(HATCH_PATS)
        task.wait(0.2)
    end
end

-- ══════════════════════════════════════════════════════════════
--  AUTO PLACE
-- ══════════════════════════════════════════════════════════════
local PLACE_REMOTES = {"PlacePack","PlaceInBase","SetPack","EquipPack","DeployPack"}
local PLACE_PATS    = {"place","equip","put","deploy"}

local function do_one_place()
    State.Placing = true
    local baseCF  = get_base_cf()
    move_to(baseCF)
    cancel_tween()
    HRP.CFrame = baseCF
    task.wait(0.1)
    for _, name in ipairs(PLACE_REMOTES) do fire_remote(name) end
    scan_fire(PLACE_PATS)
    task.wait(0.15)
end

local function auto_place_loop()
    while State.Placing do
        do_one_place()
        task.wait(0.3)
    end
    State.Placing = false
end

-- ══════════════════════════════════════════════════════════════
--  NOTIF HOOK — watches every TextLabel for tier keywords
--  fires do_one_steal immediately on match
-- ══════════════════════════════════════════════════════════════
local notif_conns = {}

local function hook_label(obj)
    if not (obj:IsA("TextLabel") or obj:IsA("TextBox")) then return end
    local function check(txt)
        if str_has_tier(txt) and State.Stealing and not steal_busy then
            local t = nearest_target()
            if t then task.spawn(do_one_steal, t) end
        end
    end
    check(obj.Text)
    local c = obj:GetPropertyChangedSignal("Text"):Connect(function()
        check(obj.Text)
    end)
    table.insert(notif_conns, c)
end

local function hook_container(container)
    for _, obj in ipairs(container:GetDescendants()) do
        hook_label(obj)
    end
    local c = container.DescendantAdded:Connect(function(obj)
        task.wait(0.03)
        hook_label(obj)
    end)
    table.insert(notif_conns, c)
end

local function setup_notif_hooks()
    local pg = LP:WaitForChild("PlayerGui")
    for _, gui in ipairs(pg:GetChildren()) do
        local skip = false
        for _, n in ipairs(GUI_ROOTS) do
            if gui.Name == n then skip = true end
        end
        if not skip then hook_container(gui) end
    end
    pg.ChildAdded:Connect(function(gui)
        local skip = false
        for _, n in ipairs(GUI_ROOTS) do
            if gui.Name == n then skip = true end
        end
        if not skip then task.wait(0.05); hook_container(gui) end
    end)
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BillboardGui") or obj:IsA("SurfaceGui") then
            hook_container(obj)
        end
    end
    Workspace.DescendantAdded:Connect(function(obj)
        if obj:IsA("BillboardGui") or obj:IsA("SurfaceGui") then
            task.wait(0.05); hook_container(obj)
        end
    end)
end
